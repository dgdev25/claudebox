13/// Builder for assembling an AGI cognitive container manifest.
14///
15/// The manifest is a META segment in the RVF file. Other segments
16/// (KERNEL_SEG, WASM_SEG, VEC_SEG, etc.) are added through the
17/// main RVF write path; this builder only handles the manifest.
18#[derive(Clone, Debug)]
19pub struct AgiContainerBuilder {
20    /// Container UUID.
21    pub container_id: [u8; 16],
22    /// Build UUID.
23    pub build_id: [u8; 16],
24    /// Pinned model identifier string.
25    pub model_id: Option<Vec<u8>>,
26    /// Governance policy (binary).
27    pub policy: Option<Vec<u8>>,
28    /// Policy hash from ADR-035 GovernancePolicy.
29    pub policy_hash: [u8; 8],
30    /// Orchestrator config (Claude Code + Claude Flow).
31    pub orchestrator_config: Option<Vec<u8>>,
32    /// MCP tool adapter registry.
33    pub tool_registry: Option<Vec<u8>>,
34    /// Agent role prompts.
35    pub agent_prompts: Option<Vec<u8>>,
36    /// Evaluation task suite.
37    pub eval_tasks: Option<Vec<u8>>,
38    /// Grading rules.
39    pub eval_graders: Option<Vec<u8>>,
40    /// Skill library.
41    pub skill_library: Option<Vec<u8>>,
42    /// Replay script.
43    pub replay_script: Option<Vec<u8>>,
44    /// Kernel boot config.
45    pub kernel_config: Option<Vec<u8>>,
46    /// Network config.
47    pub network_config: Option<Vec<u8>>,
48    /// Coherence gate config.
49    pub coherence_config: Option<Vec<u8>>,
50    /// Project instructions (CLAUDE.md).
51    pub project_instructions: Option<Vec<u8>>,
52    /// Dependency snapshot.
53    pub dependency_snapshot: Option<Vec<u8>>,
54    /// Authority level and resource budget config.
55    pub authority_config: Option<Vec<u8>>,
56    /// Target domain profile.
57    pub domain_profile: Option<Vec<u8>>,
58    /// Segment inventory.
59    pub segments: ContainerSegments,

62}

207    /// Build the manifest TLV payload (sections only, no header).
208    fn build_sections(&self) -> Vec<u8> {
209        let mut payload = Vec::new();
210
211        let mut write_section = |tag: u16, data: &[u8]| {
212            payload.extend_from_slice(&tag.to_le_bytes());
213            payload.extend_from_slice(&(data.len() as u32).to_le_bytes());
214            payload.extend_from_slice(data);
215        };
216
217        write_section(AGI_TAG_CONTAINER_ID, &self.container_id);
218        write_section(AGI_TAG_BUILD_ID, &self.build_id);
219
220        if let Some(ref mid) = self.model_id {
221            write_section(AGI_TAG_MODEL_ID, mid);
222        }
223        if let Some(ref p) = self.policy {
224            write_section(AGI_TAG_POLICY, p);
225        }
226        if let Some(ref oc) = self.orchestrator_config {
227            write_section(AGI_TAG_ORCHESTRATOR, oc);
228        }
229        if let Some(ref tr) = self.tool_registry {
230            write_section(AGI_TAG_TOOL_REGISTRY, tr);
231        }
232        if let Some(ref ap) = self.agent_prompts {
233            write_section(AGI_TAG_AGENT_PROMPTS, ap);
234        }
235        if let Some(ref et) = self.eval_tasks {
236            write_section(AGI_TAG_EVAL_TASKS, et);
237        }
238        if let Some(ref eg) = self.eval_graders {
239            write_section(AGI_TAG_EVAL_GRADERS, eg);
240        }
241        if let Some(ref sl) = self.skill_library {
242            write_section(AGI_TAG_SKILL_LIBRARY, sl);
243        }
244        if let Some(ref rs) = self.replay_script {
245            write_section(AGI_TAG_REPLAY_SCRIPT, rs);
246        }
247        if let Some(ref kc) = self.kernel_config {
248            write_section(AGI_TAG_KERNEL_CONFIG, kc);
249        }
250        if let Some(ref nc) = self.network_config {
251            write_section(AGI_TAG_NETWORK_CONFIG, nc);
252        }
253        if let Some(ref cc) = self.coherence_config {
254            write_section(AGI_TAG_COHERENCE_CONFIG, cc);
255        }
256        if let Some(ref pi) = self.project_instructions {
257            write_section(AGI_TAG_PROJECT_INSTRUCTIONS, pi);
258        }
259        if let Some(ref ds) = self.dependency_snapshot {
260            write_section(AGI_TAG_DEPENDENCY_SNAPSHOT, ds);
261        }
262        if let Some(ref ac) = self.authority_config {
263            write_section(AGI_TAG_AUTHORITY_CONFIG, ac);
264        }
265        if let Some(ref dp) = self.domain_profile {
266            write_section(AGI_TAG_DOMAIN_PROFILE, dp);
267        }
268
269        payload
270    }
271
272    /// Build the manifest: header + TLV sections.
273    pub fn build(self) -> Result<(Vec<u8>, AgiContainerHeader), ContainerError> {
274        let sections = self.build_sections();
275
276        let model_id_hash = match &self.model_id {
277            Some(mid) => seed_crypto::seed_content_hash(mid),
278            None => [0u8; 8],
279        };
280
281        let flags = self.segments.to_flags() | self.extra_flags;
282
283        let created_ns = SystemTime::now()
284            .duration_since(UNIX_EPOCH)
285            .unwrap_or_default()
286            .as_nanos() as u64;
287
288        let header = AgiContainerHeader {
289            magic: AGI_MAGIC,
290            version: 1,
291            flags,
292            container_id: self.container_id,
293            build_id: self.build_id,
294            created_ns,
295            model_id_hash,
296            policy_hash: self.policy_hash,
297        };
298
299        let mut payload = Vec::with_capacity(AGI_HEADER_SIZE + sections.len());
300        payload.extend_from_slice(&header.to_bytes());
301        payload.extend_from_slice(&sections);
302
303        // Mark manifest as present for validation.
304        let mut segs = self.segments.clone();
305        segs.manifest_present = true;
306
307        Ok((payload, header))
308    }
309
310    /// Build and sign with HMAC-SHA256.
311    pub fn build_and_sign(
312        mut self,
313        key: &[u8],
314    ) -> Result<(Vec<u8>, AgiContainerHeader), ContainerError> {
315        self.segments.crypto_present = true;
316        let (unsigned, mut header) = self.build()?;
317        header.flags |= AGI_SIGNED;
318
319        let sig = seed_crypto::sign_seed(key, &unsigned);
320        let mut signed = unsigned;
321        // Re-write the header with SIGNED flag.
322        let header_bytes = header.to_bytes();
323        signed[..AGI_HEADER_SIZE].copy_from_slice(&header_bytes);
324        signed.extend_from_slice(&sig);
325
326        Ok((signed, header))
327    }

330/// Parsed AGI container manifest with zero-copy section references.
331#[derive(Debug)]
332pub struct ParsedAgiManifest<'a> {
333    /// Parsed header.
334    pub header: AgiContainerHeader,
335    /// Model identifier string.
336    pub model_id: Option<&'a [u8]>,
337    /// Policy bytes.
338    pub policy: Option<&'a [u8]>,
339    /// Orchestrator config.
340    pub orchestrator_config: Option<&'a [u8]>,
341    /// Tool registry.
342    pub tool_registry: Option<&'a [u8]>,
343    /// Agent prompts.
344    pub agent_prompts: Option<&'a [u8]>,
345    /// Eval tasks.
346    pub eval_tasks: Option<&'a [u8]>,
347    /// Eval graders.
348    pub eval_graders: Option<&'a [u8]>,
349    /// Skill library.
350    pub skill_library: Option<&'a [u8]>,
351    /// Replay script.
352    pub replay_script: Option<&'a [u8]>,
353    /// Kernel config.
354    pub kernel_config: Option<&'a [u8]>,
355    /// Network config.
356    pub network_config: Option<&'a [u8]>,
357    /// Coherence gate config.
358    pub coherence_config: Option<&'a [u8]>,
359    /// Project instructions.
360    pub project_instructions: Option<&'a [u8]>,
361    /// Dependency snapshot.
362    pub dependency_snapshot: Option<&'a [u8]>,
363    /// Authority configuration.
364    pub authority_config: Option<&'a [u8]>,
365    /// Domain profile.

367}
368
369impl<'a> ParsedAgiManifest<'a> {
370    /// Parse a manifest from bytes.
371    pub fn parse(data: &'a [u8]) -> Result<Self, ContainerError> {
372        let header = AgiContainerHeader::from_bytes(data)
373            .map_err(|_| ContainerError::InvalidConfig("invalid header"))?;
374
375        let mut result = Self {
376            header,
377            model_id: None,
378            policy: None,
379            orchestrator_config: None,
380            tool_registry: None,
381            agent_prompts: None,
382            eval_tasks: None,
383            eval_graders: None,
384            skill_library: None,
385            replay_script: None,
386            kernel_config: None,
387            network_config: None,
388            coherence_config: None,
389            project_instructions: None,
390            dependency_snapshot: None,
391            authority_config: None,
392            domain_profile: None,
393        };
394
395        // Parse TLV sections after header.
396        let mut pos = AGI_HEADER_SIZE;
397        while pos + 6 <= data.len() {
398            let tag = u16::from_le_bytes([data[pos], data[pos + 1]]);
399            let length = u32::from_le_bytes([
400                data[pos + 2], data[pos + 3], data[pos + 4], data[pos + 5],
401            ]) as usize;
402            pos += 6;
403
404            if pos + length > data.len() {
405                break;
406            }
407
408            let value = &data[pos..pos + length];
409            match tag {
410                AGI_TAG_MODEL_ID => result.model_id = Some(value),
411                AGI_TAG_POLICY => result.policy = Some(value),
412                AGI_TAG_ORCHESTRATOR => result.orchestrator_config = Some(value),
413                AGI_TAG_TOOL_REGISTRY => result.tool_registry = Some(value),
414                AGI_TAG_AGENT_PROMPTS => result.agent_prompts = Some(value),
415                AGI_TAG_EVAL_TASKS => result.eval_tasks = Some(value),
416                AGI_TAG_EVAL_GRADERS => result.eval_graders = Some(value),
417                AGI_TAG_SKILL_LIBRARY => result.skill_library = Some(value),
418                AGI_TAG_REPLAY_SCRIPT => result.replay_script = Some(value),
419                AGI_TAG_KERNEL_CONFIG => result.kernel_config = Some(value),
420                AGI_TAG_NETWORK_CONFIG => result.network_config = Some(value),
421                AGI_TAG_COHERENCE_CONFIG => result.coherence_config = Some(value),
422                AGI_TAG_PROJECT_INSTRUCTIONS => result.project_instructions = Some(value),
423                AGI_TAG_DEPENDENCY_SNAPSHOT => result.dependency_snapshot = Some(value),
424                AGI_TAG_AUTHORITY_CONFIG => result.authority_config = Some(value),
425                AGI_TAG_DOMAIN_PROFILE => result.domain_profile = Some(value),
426                _ => {} // forward-compat: ignore unknown tags
427            }
428
429            pos += length;
430        }
431
432        Ok(result)
433    }
434
435    /// Get the model ID as a UTF-8 string.
436    pub fn model_id_str(&self) -> Option<&str> {
437        self.model_id.and_then(|b| core::str::from_utf8(b).ok())
438    }
439
440    /// Check if the manifest has all sections needed for autonomous operation.
441    pub fn is_autonomous_capable(&self) -> bool {
442        self.orchestrator_config.is_some()
443            && self.eval_tasks.is_some()
444            && self.eval_graders.is_some()
445    }
446}
